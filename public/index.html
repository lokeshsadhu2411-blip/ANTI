Copy the code below, go to your Supabase Dashboard → SQL Editor, paste and click Run: 

CREATE TABLE IF NOT EXISTS public.complaints (
    id TEXT PRIMARY KEY,
    title TEXT NOT NULL,
    category TEXT NOT NULL,
    category_label TEXT,
    description TEXT,
    severity TEXT DEFAULT 'Medium',
    department TEXT NOT NULL,
    status TEXT DEFAULT 'Reported',
    location JSONB NOT NULL DEFAULT '{}'::jsonb,
    image_url TEXT,
    resolution_proof JSONB,
    upvotes INTEGER DEFAULT 1,
    reported_at TIMESTAMPTZ DEFAULT NOW(),
    sla_hours INTEGER DEFAULT 48,
    assigned_officer JSONB,
    citizen_contact JSONB,
    timeline JSONB DEFAULT '[]'::jsonb
);

ALTER TABLE public.complaints ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public read" ON public.complaints FOR SELECT USING (true);
CREATE POLICY "Allow public insert" ON public.complaints FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow public update" ON public.complaints FOR UPDATE USING (true);
